Cleanup Dataset

ChemData_clean <- ChemData |>
  drop_na() |> # removes all rows containing an NA
  separate_wider_delim(
    cols = Tide_time,
    delim = "_",
    names = c("Tide", "Time"),
    cols_remove = FALSE) |>
  mutate(Site_Zone = paste(Site, Zone, sep = ".")) # combines Site and Zone into one column

view(ChemData_clean)

This code is from my Week 4 homework. Since I’m using the same dataset again, I’m just fixing my data again.

Filtering a Subset

ChemData_HighTide <- ChemData_clean |>

Plotting pH Data

ChemData_HighTide_pH <- ggplot(data = ChemData_HighTide, mapping = aes(x = Site_Zone,y = pH,color = Site_Zone)) +
  geom_jitter(width = 0.12,height = 0,size = 1.5,alpha = 0.7) +
  labs(x = "Site and Zone", y = "pH", color = "Site and Zone") +
  scale_color_manual(values = c("#67001f", "#b2182b", "#d6604d", "#f4a582","#4393c3", "#2166ac", "#053061")) +
  theme_classic()

Creating a Table

ChemData_HighTide |>
  group_by(Site_Zone) |>
  summarise(pH_mean = mean(pH, na.rm = TRUE)) |>
  kbl() |>
  kable_classic() |>
  row_spec(c(2, 4, 6), bold = TRUE, color = "white", background = "green") |> # highlight every other row, just for fun
  kable_styling(full_width = FALSE)
