# https://dplyr.tidyverse.org/articles/programming.html

WaterQualityStats <- function(UlvaFieldData, variable) {
  
  UlvaFieldData |> 
    group_by(Site_ID) |> 
    summarise(min = min({{variable}}, na.rm = TRUE),
              max = max({{variable}}, na.rm = TRUE),
              mean = mean({{variable}}, na.rm = TRUE))

}
